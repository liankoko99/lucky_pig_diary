        function CalendarView({ entries, currentDate, setCurrentDate, onSelectDate }) {

            const monthNames = ["一月", "二月", "三月", "四月", "五月", "六月", "七月", "八月", "九月", "十月", "十一月", "十二月"];

            const year = currentDate.getFullYear();

            const month = currentDate.getMonth();


            const daysInMonth = new Date(year, month + 1, 0).getDate();

            const firstDay = new Date(year, month, 1).getDay();

            

            const days = [];

            for (let i = 0; i < firstDay; i++) days.push(null);

            for (let d = 1; d <= daysInMonth; d++) days.push(new Date(year, month, d));


            return (

                <div className="animate-fade-in">

                    <div className="flex justify-between items-center mb-6">

                        <h2 className="text-xl font-bold text-[#1A1A1A]">

                        </h2>

                        <div className="flex gap-3">

                            <button onClick={() => setCurrentDate(new Date(year, month - 1, 1))} className="w-10 h-10 border-2 border-[#1A1A1A] bg-white rounded-full flex items-center justify-center hover:bg-[#FCF5CA] active:scale-95 transition-all shadow-[2px_2px_0px_#1A1A1A]">

                        {['日', '一', '二', '三', '四', '五', '六'].map(d => (

        function App() {

            const [view, setView] = useState('calendar'); // 'calendar', 'entry', 'share'

            const [user, setUser] = useState(null);

            const [roomId, setRoomId] = useState(() => localStorage.getItem('gratitude_room_id') || '');

            const [entries, setEntries] = useState(() => {

                const saved = localStorage.getItem('gratitude_entries');

                return saved ? JSON.parse(saved) : {};

            });

            const [isSyncing, setIsSyncing] = useState(false);

            

            const [currentDate, setCurrentDate] = useState(new Date());

            const [selectedDate, setSelectedDate] = useState(() => {

                const today = new Date();

                return `${today.getFullYear()}-${String(today.getMonth() + 1).padStart(2, '0')}-${String(today.getDate()).padStart(2, '0')}`;

            });


            // 1. 處理 Firebase 匿名登入

            useEffect(() => {

                if (!auth) return;

                const initAuth = async () => {

                    try {

                        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

                            await signInWithCustomToken(auth, __initial_auth_token);

                        } else {

                            await signInAnonymously(auth);

                        }

                    } catch (err) {

                        console.error("Auth error", err);

                    }

                };

                initAuth();

                const unsubscribe = onAuthStateChanged(auth, setUser);

                return () => unsubscribe();

            }, []);


            // 2. 處理資料同步 (如果設定了 roomId)

            useEffect(() => {

                if (!user) return;

                

                if (roomId) {

                    setIsSyncing(true);

                    const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'shared_diaries', roomId);

                    

                    const unsubscribe = onSnapshot(docRef, (docSnap) => {

                        if (docSnap.exists()) {

                            const cloudData = docSnap.data().entries || {};

                            setEntries(cloudData);

                            localStorage.setItem('gratitude_entries', JSON.stringify(cloudData));

                        } else {

                            // 如果這個分享碼還沒有資料，就建立它

                            setDoc(docRef, { entries: {} }, { merge: true });

                        }

                        setIsSyncing(false);

                    }, (error) => {

                        console.error("Sync error:", error);

                        setIsSyncing(false);

                    });

                    

                    return () => unsubscribe();

                } else {

                    // 單機模式：讀取本地資料

                    const saved = localStorage.getItem('gratitude_entries');

                    if (saved) setEntries(JSON.parse(saved));

                }

            }, [user, roomId]);


            const handleSave = async (date, data) => {

                const newEntries = { ...entries, [date]: data };

                setEntries(newEntries); // 畫面先即時更新

                

                // 單機儲存

                localStorage.setItem('gratitude_entries', JSON.stringify(newEntries));


                // 雲端儲存

                if (roomId && user) {

                    try {

                        const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'shared_diaries', roomId);

                        await setDoc(docRef, { entries: newEntries }, { merge: true });

                    } catch (err) {

                        console.error("Cloud save failed", err);

                    }

                }

            };


            return (

                <div className="min-h-screen flex justify-center p-3 sm:p-8 relative">

                    <div className="noise-overlay" />

                    <div className="w-full max-w-lg min-h-[90vh] hand-drawn-border relative overflow-hidden flex flex-col p-5 sm:p-10 shadow-[10px_10px_0px_rgba(0,0,0,0.15)] bg-[#B4E082]">

                        <header className="mb-6 relative z-10 flex justify-between items-start">

                            <div>

                                <div className="flex items-center gap-2 mb-2">

                                    <div className="p-1.5 bg-[#1A1A1A] rounded-full text-white">

                                        <PenToolIcon />

                                    </div>

                                    <span className="text-[11px] font-bold tracking-[0.2em] uppercase text-[#1A1A1A]/60">lucky pig pig</span>

                                </div>

                                <h1 className="text-3xl font-bold tracking-tight text-[#1A1A1A]">

                                    {view === 'calendar' ? '感恩日記' : view === 'share' ? '交換日記' : '今日小確幸'}

                                </h1>

                            </div>

                            

                            {/* 連線按鈕 / 狀態列 */}

                            <button 

                                onClick={() => setView('share')}

                                className={`flex items-center gap-1.5 px-3 py-1.5 rounded-full border-2 border-[#1A1A1A] font-bold text-[10px] transition-all shadow-[2px_2px_0px_#1A1A1A] active:translate-x-0.5 active:translate-y-0.5 active:shadow-none ${roomId ? 'bg-[#FCF5CA] text-[#1A1A1A]' : 'bg-white text-[#1A1A1A]/60'}`}

                            >

                                <ShareIcon />

                                {roomId ? '已連線' : '邀請朋友'}

                            </button>

                        </header>


                        <main className="flex-1 z-10 overflow-y-auto overflow-x-hidden pb-4">

                            {view === 'calendar' ? (

                                <CalendarView entries={entries} currentDate={currentDate} setCurrentDate={setCurrentDate} onSelectDate={(d) => { setSelectedDate(d); setView('entry'); }} />

                            ) : view === 'share' ? (

                                <ShareView roomId={roomId} setRoomId={setRoomId} onBack={() => setView('calendar')} />

                            ) : (

                                <EntryView date={selectedDate} existingData={entries[selectedDate]} onBack={() => setView('calendar')} onSave={(data) => handleSave(selectedDate, data)} />

                            )}

                        </main>

                    </div>

                </div>

            );

        }


        const root = ReactDOM.createRoot(document.getElementById('root'));

        root.render(<App />);

    </script>
